如何連結您的私人 Google 試算表與雲端硬碟？

透過免費的 Google Apps Script (GAS)，您可以將資料直接同步儲存在您自己的 Google 試算表中，換手機或電腦只要貼上網址就能一鍵還原！

📋 3 分鐘設定教學與 Apps Script 程式碼
  1. 前往您的 Google 雲端硬碟，新增一個空白的「Google 試算表」（可命名為：產險客戶資料庫）。
  2. 在試算表上方選單點擊「擴充功能」→「Apps Script」。
  3. 刪除編輯器內原有程式碼，貼上以下程式碼並按 Ctrl+S 儲存。
  4. 點擊右上角「部署」→「新增部署作業」，類型選「網頁應用程式」。
  5. 執行身分選「我」，誰可以存取選「所有人」，點擊部署並授權後，複製產生的 網頁應用程式網址 貼到上方欄位即可！
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getRange("A1").getValue();
  if (!data) {
    data = JSON.stringify({ customers: [], companies: [] });
  }
  return ContentService.createTextOutput(data)
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var jsonStr = e.postData.contents;
    var parsed = JSON.parse(jsonStr);
    
    // 1. 將完整 JSON 備份於 A1 儲存格供系統精準還原
    sheet.getRange("A1").setValue(jsonStr);
    sheet.getRange("B1").setValue("最後同步時間：" + new Date().toLocaleString("zh-TW"));
    
    // 2. 同時在下方產生可讀的客戶明細表方便直接瀏覽
    if (sheet.getLastRow() > 2) {
      sheet.getRange(3, 1, sheet.getLastRow() - 2, 8).clearContent();
    }
    var headers = [["客戶姓名", "身分證/統編", "車牌號碼", "產險公司", "保險起始日", "保險到期日", "聯絡電話", "備註"]];
    sheet.getRange(3, 1, 1, 8).setValues(headers).setFontWeight("bold");
    
    if (parsed.customers && parsed.customers.length > 0) {
      var rows = parsed.customers.map(function(c) {
        return [c.name, c.idNumber, c.plate, c.company, c.startDate, c.endDate, c.phone || "", c.notes || ""];
      });
      sheet.getRange(4, 1, rows.length, 8).setValues(rows);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: parsed.customers.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
